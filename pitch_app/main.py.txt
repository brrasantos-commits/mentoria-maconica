from pathlib import Path
from io import BytesIO
import json

from fastapi import FastAPI, Request, Form, File, UploadFile, HTTPException, Query

from pitch_app.services.evaluation_service import evaluate_submission
from pitch_app.services.exceptions import AppError
from pitch_app.services.job_store import create_job, get_job, update_job

BASE_DIR = Path(__file__).resolve().parent
TEMPLATES_DIR = BASE_DIR / "templates"
STATIC_DIR = BASE_DIR / "static"
MATERIALS_DIR = BASE_DIR / "materials"
MATERIALS_META_FILE = BASE_DIR / "materials.json"
SESSION_SECRET = "change-this-secret-in-production"

STATIC_DIR.mkdir(parents=True, exist_ok=True)
MATERIALS_DIR.mkdir(parents=True, exist_ok=True)
(STATIC_DIR / "css").mkdir(parents=True, exist_ok=True)

app = FastAPI(title="Sales Pitch AI V2.3")
app.add_middleware(SessionMiddleware, secret_key=SESSION_SECRET)
templates = Jinja2Templates(directory=str(TEMPLATES_DIR))
app.mount("/static", StaticFiles(directory=str(STATIC_DIR)), name="static")

INDUSTRY_OPTIONS = ["Todas", "Varejo", "Saúde", "Finanças", "Tecnologia", "Educação", "Indústria"]
SOLUTION_OPTIONS = ["Todas", "Software", "Serviços", "Consultoria", "Hardware", "Plataforma"]


def _load_material_metadata() -> list[dict]:
    if not MATERIALS_META_FILE.exists():
        return []
    try:
        data = json.loads(MATERIALS_META_FILE.read_text(encoding="utf-8"))
        return data if isinstance(data, list) else []
    except Exception:
        return []


def _material_title(name: str) -> str:
    return Path(name).stem.replace("_", " ").replace("-", " ").title()


def _guess_kind(ext: str) -> str:
    ext = ext.lower().lstrip(".")
    if ext == "pdf":
        return "pdf"
    if ext in {"mp4", "m4v", "webm", "mov", "avi", "mkv"}:
        return "video"
    return "file"


def _normalize_material(item: dict) -> dict:
    filename = item.get("filename") or item.get("file") or ""
    path = MATERIALS_DIR / filename
    ext = Path(filename).suffix.lower().lstrip(".")
    return {
        "slug": item.get("slug") or Path(filename).stem,
        "title": item.get("title") or _material_title(filename),
        "path": f"/material/{filename}",
        "type": item.get("type") or ext,
        "kind": item.get("kind") or _guess_kind(ext),
        "filename": filename,
        "industry": item.get("industry") or "Não informado",
        "solution": item.get("solution") or "Não informado",
        "description": item.get("description") or "",
        "sort": item.get("sort", 0),
        "exists": path.exists(),
    }


def _list_materials():
    items = [_normalize_material(item) for item in _load_material_metadata()]
    items = [m for m in items if m["filename"] and m["exists"]]
    items.sort(key=lambda x: (x.get("sort", 0), x["title"].lower()))
    return items


def _filter_materials(materials, industry=None, solution=None):
    if industry and industry != "Todas":
        materials = [m for m in materials if m["industry"] == industry]
    if solution and solution != "Todas":
        materials = [m for m in materials if m["solution"] == solution]
    return materials


def _selected_materials_from_session(request: Request):
    selected = request.session.get("selected_materials", [])
    materials = _list_materials()
    by_filename = {m["filename"]: m for m in materials}
    return [by_filename[name] for name in selected if name in by_filename]


async def _material_to_uploadfile(material: dict) -> UploadFile:
    file_path = MATERIALS_DIR / material["filename"]
    data = file_path.read_bytes()
    return UploadFile(filename=material["filename"], file=BytesIO(data))


@app.get("/", response_class=HTMLResponse)
async def home():
    return RedirectResponse(url="/estudo", status_code=303)


@app.get("/estudo", response_class=HTMLResponse)
async def study_index(
    request: Request,
    industry: str = Query("Todas"),
    solution: str = Query("Todas"),
):
    materials = _filter_materials(_list_materials(), industry, solution)
    selected = set(request.session.get("selected_materials", []))
    return templates.TemplateResponse("study_index.html", {
        "request": request,
        "materials": materials,
        "selected": selected,
        "industry_options": INDUSTRY_OPTIONS,
        "solution_options": SOLUTION_OPTIONS,
        "current_industry": industry,
        "current_solution": solution,
    })


@app.get("/material/{filename}")
async def open_material(filename: str):
    path = MATERIALS_DIR / filename
    if not path.exists() or not path.is_file():
        raise HTTPException(status_code=404, detail="Material não encontrado")

    ext = path.suffix.lower()
    media_type = (
        "application/pdf" if ext == ".pdf"
        else "video/mp4" if ext == ".mp4"
        else "video/webm" if ext == ".webm"
        else None
    )

    headers = {"Content-Disposition": f'inline; filename="{path.name}"'}
    return FileResponse(path, media_type=media_type, headers=headers)


@app.get("/material/{filename}/view", response_class=HTMLResponse)
async def material_view(request: Request, filename: str):
    path = MATERIALS_DIR / filename
    if not path.exists() or not path.is_file():
        raise HTTPException(status_code=404, detail="Material não encontrado")

    material = {
        "filename": path.name,
        "title": _material_title(path.name),
        "type": path.suffix.lower().lstrip("."),
        "kind": _guess_kind(path.suffix),
    }
    return templates.TemplateResponse("study_reader.html", {
        "request": request,
        "material": material,
    })


@app.post("/estudo/concluir")
async def study_complete(request: Request, selected_materials: list[str] = Form(default=[])):
    request.session["selected_materials"] = selected_materials
    return RedirectResponse(url="/pitch", status_code=303)


@app.get("/pitch", response_class=HTMLResponse)
async def pitch_form(request: Request):
    materials = _selected_materials_from_session(request)
    return templates.TemplateResponse("pitch_form.html", {
        "request": request,
        "materials": materials,
    })


@app.get("/api/jobs/{job_id}")
async def job_status(job_id: str):
    job = get_job(job_id)
    if not job:
        return JSONResponse(status_code=404, content={"detail": "Job não encontrado"})
    return JSONResponse(content=job)


@app.post("/analyze", response_class=HTMLResponse)
async def analyze(request: Request, seller_name: str = Form(...), video: UploadFile = File(...)):
    selected_names = request.session.get("selected_materials", [])
    available = {m["filename"]: m for m in _list_materials()}
    chosen = [available[name] for name in selected_names if name in available]

    if not chosen:
        raise HTTPException(status_code=400, detail="Selecione pelo menos um material na etapa de estudo.")

    job_id = create_job(seller_name=(seller_name or "").strip(), video_name=video.filename or "")

    try:
        material_files = [await _material_to_uploadfile(m) for m in chosen]
        result = evaluate_submission(job_id=job_id, seller_name=seller_name, video=video, materials=material_files)
        return templates.TemplateResponse("result.html", {"request": request, **result})
    except AppError as exc:
        update_job(job_id, stage="error", progress=100, message=exc.message, status="error")
        raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc
    except Exception as exc:
        update_job(job_id, stage="error", progress=100, message=str(exc), status="error")
        raise HTTPException(status_code=500, detail=f"Erro interno ao analisar o pitch: {exc}") from exc

def _guess_type(filename: str) -> str:

    return MATERIALS_DIR / filename


def _save_upload(upload: UploadFile, destination: Path):
    with destination.open("wb") as buffer:
        copyfileobj(upload.file, buffer)

async def lifespan(app: FastAPI):
    Base.metadata.create_all(bind=engine)
    yield

def _get_materials_from_db(db: Session, industry: str = "Todas", solution: str = "Todas"):
    query = db.query(Material).filter(Material.active == True)

    if industry and industry != "Todas":
        query = query.filter(Material.industry == industry)

    if solution and solution != "Todas":
        query = query.filter(Material.solution == solution)

    return query.order_by(Material.sort_order.asc(), Material.id.asc()).all()

@app.get("/estudo", response_class=HTMLResponse)
def study_index(
    request: Request,
    industry: str = Query("Todas"),
    solution: str = Query("Todas"),
    db: Session = Depends(get_db),
):
    materials = _get_materials_from_db(db, industry=industry, solution=solution)

    industries = ["Todas"] + [
        row[0] for row in db.query(Material.industry)
        .filter(Material.active == True)
        .distinct()
        .order_by(Material.industry.asc())
        .all()
    ]

    solutions = ["Todas"] + [
        row[0] for row in db.query(Material.solution)
        .filter(Material.active == True)
        .distinct()
        .order_by(Material.solution.asc())
        .all()
    ]

    return request.app.state.templates.TemplateResponse(
        "study_index.html",
        {
            "request": request,
            "materials": materials,
            "industry_options": industries,
            "solution_options": solutions,
            "current_industry": industry,
            "current_solution": solution,
        },
    )

        "study_reader.html",
        {
            "request": request,
            "material": material,
        },
    )


@app.get("/material/{material_id}")
def open_material(material_id: int, db: Session = Depends(get_db)):
    material = db.query(Material).filter(Material.id == material_id, Material.active == True).first()
    if not material:
        raise HTTPException(status_code=404, detail="Material não encontrado")

    path = _material_file_path(material.filename)
    if not path.exists():
        raise HTTPException(status_code=404, detail="Arquivo não encontrado")

    media_type = (
        "application/pdf" if material.file_type == "pdf"
        else f"video/{material.file_type}" if material.file_type in {"mp4", "webm", "mov", "avi", "mkv"}
        else None
    )

    return FileResponse(path, media_type=media_type, filename=material.filename)

@app.get("/estudo", response_class=HTMLResponse)
def study_index(request: Request):
    db = SessionLocal()
    try:
        rows = db.execute(text("""
            SELECT id, title, filename, file_type, industry, solution, sort_order
            FROM materials
            WHERE active = 1
            ORDER BY sort_order ASC, id ASC
        """)).fetchall()

        materials = [
            {
                "id": row.id,
                "title": row.title,
                "filename": row.filename,
                "file_type": row.file_type,
                "industry": row.industry,
                "solution": row.solution,
                "sort_order": row.sort_order,
            }
            for row in rows
        ]

        industries = ["Todas"] + [r[0] for r in db.execute(text("""
            SELECT DISTINCT industry
            FROM materials
            WHERE active = 1
            ORDER BY industry ASC
        """)).fetchall()]

        solutions = ["Todas"] + [r[0] for r in db.execute(text("""
            SELECT DISTINCT solution
            FROM materials
            WHERE active = 1
            ORDER BY solution ASC
        """)).fetchall()]
    finally:
        db.close()

    return request.app.state.templates.TemplateResponse(
        "study_index.html",

from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from starlette.middleware.sessions import SessionMiddleware

from pitch_app.admin_routes import router as admin_router
from pitch_app.db import init_db
from pitch_app.services.evaluation_service import evaluate_submission
from pitch_app.services.exceptions import AppError
from pitch_app.services.job_store import create_job, get_job, update_job

BASE_DIR = Path(__file__).resolve().parent
TEMPLATES_DIR = BASE_DIR / "templates"
STATIC_DIR = BASE_DIR / "static"
MATERIALS_DIR = BASE_DIR / "materials"

STATIC_DIR.mkdir(parents=True, exist_ok=True)
MATERIALS_DIR.mkdir(parents=True, exist_ok=True)
(STATIC_DIR / "css").mkdir(parents=True, exist_ok=True)

app = FastAPI(title="Sales Pitch AI V2.3")
app.add_middleware(SessionMiddleware, secret_key="change-this-secret-in-production")
templates = Jinja2Templates(directory=str(TEMPLATES_DIR))
app.mount("/static", StaticFiles(directory=str(STATIC_DIR)), name="static")
app.mount("/materials", StaticFiles(directory=str(MATERIALS_DIR)), name="materials")
app.include_router(admin_router)


@app.on_event("startup")
def on_startup():
    init_db()


def _list_materials():
    items = []
    if not MATERIALS_DIR.exists():
        return items
    for p in sorted(MATERIALS_DIR.iterdir()):
        if p.is_file() and not p.name.startswith("."):
            items.append(
                {
                    "slug": p.stem,
                    "title": p.stem.replace("_", " ").replace("-", " ").title(),
                    "path": f"/materials/{p.name}",
                    "type": p.suffix.lower().lstrip("."),
                    "filename": p.name,
                }
            )
    return items


@app.get("/", response_class=HTMLResponse)
async def home():
    return RedirectResponse(url="/estudo", status_code=303)


@app.get("/estudo", response_class=HTMLResponse)
async def study_index(request: Request, industry: str = "all", solution: str = "all"):
    materials = _list_materials()

    if industry != "all":
        materials = [m for m in materials if m.get("industry") == industry]
    if solution != "all":
        materials = [m for m in materials if m.get("solution") == solution]

    industry_options = sorted({m.get("industry") for m in _list_materials() if m.get("industry")})
    solution_options = sorted({m.get("solution") for m in _list_materials() if m.get("solution")})

    return templates.TemplateResponse(
        "index.html",
        {
            "request": request,
            "materials": materials,
            "industry_options": industry_options,
            "solution_options": solution_options,
            "current_industry": industry,
            "current_solution": solution,
        },
    )


@app.get("/estudo/{slug}", response_class=HTMLResponse)
async def study_material(request: Request, slug: str):
    materials = _list_materials()
    material = next((m for m in materials if m["slug"] == slug), None)
    if not material:
        raise HTTPException(status_code=404, detail="Material não encontrado")

    return templates.TemplateResponse(
        "study_material.html",
        {"request": request, "material": material},
    )

@app.get("/pitch", response_class=HTMLResponse)
async def pitch_form(request: Request):
    materials = _list_materials()
    return templates.TemplateResponse(
        "pitch_form.html",
        {
            "request": request,
            "materials": materials,
        },
    )


@app.get("/api/jobs/{job_id}")
async def job_status(job_id: str):
    job = get_job(job_id)
    if not job:
        return JSONResponse(status_code=404, content={"detail": "Job não encontrado"})
    return JSONResponse(content=job)


@app.post("/analyze", response_class=HTMLResponse)
async def analyze(
    request: Request,
    seller_name: str = Form(...),
    video: UploadFile = File(...),
    materials: list[str] = Form(...),
):
    job_id = create_job(seller_name=(seller_name or "").strip(), video_name=video.filename or "")
    try:
        result = evaluate_submission(
            job_id=job_id,
            seller_name=seller_name,
            video=video,
            materials=materials,
        )
        return templates.TemplateResponse("result.html", {"request": request, **result})
    except AppError as exc:
        update_job(job_id, stage="error", progress=100, message=exc.message, status="error")
        raise HTTPException(status_code=exc.status_code, detail=exc.message) from exc
    except Exception as exc:
        update_job(job_id, stage="error", progress=100, message=str(exc), status="error")
        raise HTTPException(status_code=500, detail=f"Erro interno ao analisar o pitch: {exc}") from exc

def _list_materials(industry: str = "all", solution: str = "all"):
    db = SessionLocal()
    try:
        query = text("""
            SELECT id, title, filename, file_type, industry, solution, description, sort_order, active
            FROM materials
            ORDER BY sort_order ASC, id ASC
        """)
        rows = db.execute(query).fetchall()

        materials = [
            {
                "id": r.id,
                "slug": Path(r.filename).stem,
                "title": r.title,
                "path": f"/materials/{r.filename}",
                "type": (r.file_type or Path(r.filename).suffix.lower().lstrip(".")),
                "filename": r.filename,
                "industry": r.industry,
                "solution": r.solution,
                "description": r.description,
                "sort_order": r.sort_order,
                "active": bool(r.active),
            }
            for r in rows
            if bool(r.active)
        ]

        if industry != "all":
            materials = [m for m in materials if m.get("industry") == industry]
        if solution != "all":
            materials = [m for m in materials if m.get("solution") == solution]

        return materials
    finally:
        db.close()


def _filter_options():
    db = SessionLocal()
    try:
        rows = db.execute(text("""
            SELECT DISTINCT industry, solution
            FROM materials
            WHERE active = 1
        """)).fetchall()
        industry_options = sorted({r.industry for r in rows if r.industry})
        solution_options = sorted({r.solution for r in rows if r.solution})
        return industry_options, solution_options
    finally:
        db.close()

def _list_materials(industry: str = "all", solution: str = "all"):
    db = SessionLocal()
    try:
        rows = db.execute(text("""
            SELECT id, title, filename, file_type, industry, solution, description, sort_order, active
            FROM materials
            ORDER BY sort_order ASC, id ASC
        """)).fetchall()

        materials = []
        for r in rows:
            if not bool(r.active):
                continue

            item = {
                "id": r.id,
                "slug": f"{r.id}",
                "title": r.title,
                "filename": r.filename,
                "type": (r.file_type or Path(r.filename).suffix.lower().lstrip(".")),
                "industry": r.industry,
                "solution": r.solution,
                "description": r.description,
                "sort_order": r.sort_order,
                "active": bool(r.active),
                "path": f"/materials/{r.filename}",
            }

            if industry != "all" and item["industry"] != industry:
                continue
            if solution != "all" and item["solution"] != solution:
                continue

            materials.append(item)

        return materials
    finally:
        db.close()


def _filter_options():
    db = SessionLocal()
    try:
        rows = db.execute(text("""
            SELECT DISTINCT industry, solution
            FROM materials
            WHERE active = 1
        """)).fetchall()
        industry_options = sorted({r.industry for r in rows if r.industry})
        solution_options = sorted({r.solution for r in rows if r.solution})
        return industry_options, solution_options
    finally:
        db.close()

    db = SessionLocal()
    try:
        row = db.execute(text("""
            SELECT id, title, filename, file_type, industry, solution, description, sort_order, active
            FROM materials
            WHERE id = :id AND active = 1
        """), {"id": material_id}).fetchone()

        if not row:
            raise HTTPException(status_code=404, detail="Material não encontrado")

        material = {
            "id": row.id,
            "slug": f"{row.id}",
            "title": row.title,
            "filename": row.filename,
            "type": (row.file_type or Path(row.filename).suffix.lower().lstrip(".")),
            "industry": row.industry,
            "solution": row.solution,
            "description": row.description,
            "sort_order": row.sort_order,
            "active": bool(row.active),
            "path": f"/materials/{row.filename}",
        }
    finally:
        db.close()

    return templates.TemplateResponse(
        "study_material.html",
        {"request": request, "material": material},
    )
